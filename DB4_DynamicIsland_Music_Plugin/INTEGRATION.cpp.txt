// 1) COPY the Plugins/DB4MusicIsland directory to:
//      <AstonDB4 Unreal Project>/Plugins/DB4MusicIsland/
//
// 2) Add the plugin to AstonDB4.uproject (existing Plugins array):
//      {"Name": "DB4MusicIsland", "Enabled": true}
//
// 3) Add the module dependency in Source/AstonDB4/AstonDB4.Build.cs:
//      PublicDependencyModuleNames.Add("DB4MusicIsland");
//
// 4) Add this include in Source/AstonDB4/DB4GameMode.cpp:
#include "DB4MusicIslandLibrary.h"
//
// 5) At the very END of ADB4GameMode::StartPlay(), after Super::StartPlay():
if (APlayerController* LocalPlayer = GetWorld()->GetFirstPlayerController())
{
    UDB4MusicIslandLibrary::ShowMusicIsland(LocalPlayer, TEXT("http://127.0.0.1:8766/"));
}
